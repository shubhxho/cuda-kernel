cmake_minimum_required(VERSION 3.24)
project(decode_kernels VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_CUDA "Build GPU kernels and cuBLAS benchmark" ON)
enable_testing()
add_executable(host_tests tests/host_tests.cpp)
target_include_directories(host_tests PRIVATE include)
target_compile_options(host_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME host_contract COMMAND host_tests)
if(ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12.0 REQUIRED)
  add_executable(decode_bench src/benchmark.cu)
  target_include_directories(decode_bench PRIVATE include)
  target_link_libraries(decode_bench PRIVATE CUDA::cudart CUDA::cublas)
  set_target_properties(decode_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(decode_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
  add_test(NAME gpu_correctness COMMAND decode_bench --suite correctness --repeats 5 --samples 3)
endif()
